### BUG-003 — 文章详情页 XSS 防护

**修改文件**：[src/app/insights/[slug]/page.tsx](../co-website/src/app/insights/[slug]/page.tsx)

**改动**：
1. 安装 `sanitize-html` 包（替代 `dompurify` + `jsdom`，后者在 Next.js 服务端有类型兼容问题）
2. 将 Markdown → HTML 的转换逻辑从 JSX 中抽取到组件函数体中
3. 在 `dangerouslySetInnerHTML` 之前用 `sanitizeHtml()` 清洗 HTML，移除所有脚本、事件处理器等危险内容

```tsx
import sanitizeHtml from 'sanitize-html'

// 先转换 Markdown → HTML
const rawHtml = article.content
  .replace(/^### (.*)$/gm, '<h3>...</h3>')
  // ... 其他正则替换

// 再用 sanitize-html 清洗
const sanitizedHtml = sanitizeHtml(rawHtml, {
  allowedTags: ['h1', 'h2', 'h3', 'strong', 'em', 'a', 'img', ...],
  allowedAttributes: {
    a: ['href', 'target', 'rel', 'class'],
    img: ['src', 'alt', 'class', 'loading'],
    '*': ['class'],
  },
})
```

**新增依赖**：`sanitize-html`、`@types/sanitize-html`

### BUG-004 — CXRAnalyzer 定时器内存泄漏

**修改文件**：[src/components/features/CXRAnalyzer.tsx](../co-website/src/components/features/CXRAnalyzer.tsx)

**改动**：
```tsx
// 新增
import { useState, useCallback, useRef, useEffect } from 'react'

const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
const mountedRef = useRef(true)

useEffect(() => {
  mountedRef.current = true
  return () => {
    mountedRef.current = false
    if (intervalRef.current) {
      clearInterval(intervalRef.current)
      intervalRef.current = null
    }
  }
}, [])

const pollStatus = async (id: string) => {
  // 启动新轮询前清除旧定时器
  if (intervalRef.current) clearInterval(intervalRef.current)
  intervalRef.current = setInterval(async () => {
    // 所有 setState 前检查 mountedRef.current
    if (mountedRef.current) setStatus('done')
  }, 2000)
}
```
